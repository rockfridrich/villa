# Villa SDK - React Integration

> Passkey authentication for React apps. Works with Vite, Create React App, Remix.

## Install

```bash
npm install @rockfridrich/villa-sdk @rockfridrich/villa-sdk-react viem zod
```

## Simple API (Recommended)

No provider needed. Just use the hook:

```tsx
import { useVilla, VillaButton } from '@rockfridrich/villa-sdk-react'

function App() {
  const { user, signOut, isLoading } = useVilla()

  if (isLoading) return <div>Loading...</div>

  if (!user) {
    return (
      <VillaButton 
        onSignIn={(user) => console.log('Welcome', user.nickname)} 
      />
    )
  }

  return (
    <div>
      <img src={user.avatar} alt="" />
      <p>@{user.nickname}</p>
      <p>{user.address}</p>
      <button onClick={signOut}>Sign Out</button>
    </div>
  )
}
```

## useVilla Hook

```typescript
const { 
  user,           // VillaUser | null
  signIn,         // () => Promise<VillaUser>
  signOut,        // () => void

```

## Provider Pattern (Advanced)

For apps needing custom configuration:

```tsx
import { 
  VillaProvider, 
  VillaAuth, 
  useIdentity, 
  useAuth 
} from '@rockfridrich/villa-sdk-react'

function App() {
  return (
    <VillaProvider config={{ appId: 'my-app', network: 'base' }}>
      <AuthenticatedApp />
    </VillaProvider>
  )
}

function AuthenticatedApp() {
  const identity = useIdentity()
  const { signOut, isLoading } = useAuth()

  if (!identity) {
    return <VillaAuth onComplete={(r) => r.success && console.log(r.identity)} />
  }

  return (
    <div>
      <h1>Welcome, @{identity.nickname}!</h1>
      <button onClick={signOut}>Sign Out</button>
    </div>
  )
}
```

## Protected Route Pattern

```tsx
import { useVilla } from '@rockfridrich/villa-sdk-react'
import { Navigate } from 'react-router-dom'

function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const { user, isLoading } = useVilla()
  
  if (isLoading) return <div>Loading...</div>
  if (!user) return <Navigate to="/login" replace />
  
  return <>{children}</>
}

// Usage
<Route path="/dashboard" element={
  <ProtectedRoute>
    <Dashboard />
  </ProtectedRoute>
} />
```

## Auth State Listener

```tsx
import { useEffect } from 'react'
import { villa } from '@rockfridrich/villa-sdk'

function AuthListener() {
  useEffect(() => {
    const unsubscribe = villa.onAuthChange((user) => {
      if (user) {
        console.log('Signed in:', user.nickname)
        analytics.identify(user.address)
      } else {
        console.log('Signed out')
        analytics.reset()
      }
    })
    return unsubscribe
  }, [])

  return null
}
```

## TypeScript Types

```typescript
interface VillaUser {
  address: `0x${string}`
  nickname: string
  avatar: string  // URL
}

interface Identity {
  address: `0x${string}`
  nickname: string
  avatar: AvatarConfig
}

interface AvatarConfig {
  style: 'adventurer' | 'avataaars' | 'bottts' | 'thumbs'
  seed: string
}

type VillaAuthResult = 
  | { success: true; identity: Identity }
  | { success: false; error: string; code: SignInErrorCode }

type SignInErrorCode = 
  | 'CANCELLED'      // User closed
  | 'TIMEOUT'        // Too slow
  | 'NETWORK_ERROR'  // Connection failed
  | 'AUTH_ERROR'     // General error
```

## Error Handling

```tsx
function LoginPage() {
  const { signIn } = useVilla()
  const [error, setError] = useState<string | null>(null)

  const handleSignIn = async () => {
    try {
      await signIn()
    } catch (err) {
      if (err.message === 'CANCELLED') {
        // User closed - expected, ignore
        return
      }
      setError('Sign in failed. Please try again.')
    }
  }

  return (
    <div>
      {error && <p className="error">{error}</p>}
      <button onClick={handleSignIn}>Sign In</button>
    </div>
  )
}
```
